local Fluent = loadstring(game:HttpGet("https://github.com/dawid-scripts/Fluent/releases/latest/download/main.lua"))()
local SaveManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/master/Addons/SaveManager.lua"))()
local InterfaceManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/master/Addons/InterfaceManager.lua"))()

local Window = Fluent:CreateWindow({
    Title = "PDZ Hub",
    SubTitle = "not share",
    TabWidth = 160,
    Size = UDim2.fromOffset(400, 309),
    Acrylic = true,
    Theme = "Dark",
    MinimizeKey = Enum.KeyCode.RightControl
})

-- Nút bật/tắt UI
local ToggleUI = Instance.new("ScreenGui")
local Button = Instance.new("TextButton")

ToggleUI.Name = "ToggleUI"
ToggleUI.Parent = game.CoreGui

Button.Parent = ToggleUI
Button.Size = UDim2.new(0, 50, 0, 50)
Button.Position = UDim2.new(0, 15, 0.5, -25)
Button.BackgroundColor3 = Color3.fromRGB(20,20,20)
Button.Text = "UI"
Button.TextColor3 = Color3.new(1,1,1)
Button.TextScaled = true
Button.BorderSizePixel = 0
Button.Active = true
Button.Draggable = true

local UICorner = Instance.new("UICorner", Button)
UICorner.CornerRadius = UDim.new(1,0)

-- Toggle Fluent Window
local Visible = true

Button.MouseButton1Click:Connect(function()
    Visible = not Visible
    
    game:GetService("CoreGui").Fluent.Visible = Visible
    -- Nếu lỗi thì đổi "Fluent" thành tên ScreenGui của bạn
end)
