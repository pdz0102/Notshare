local Fluent = loadstring(game:HttpGet("https://github.com/dawid-scripts/Fluent/releases/latest/download/main.lua"))()
local SaveManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/master/Addons/SaveManager.lua"))()
local InterfaceManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/master/Addons/InterfaceManager.lua"))()

local Window = Fluent:CreateWindow({
    Title = "PDZ Hub",
    SubTitle = "not share",
    TabWidth = 160,
    Size = UDim2.fromOffset(400, 309),
    Acrylic = true,
    Theme = "Dark",
    MinimizeKey = Enum.KeyCode.RightControl
})

-- ========================================
-- LẤY FRAME CHÍNH CỦA FLUENT
-- Fluent lưu GUI trong Window.GUI hoặc Window.Gui
-- ========================================
task.wait(0.5)

local FluentFrame = nil

-- Thử lấy trực tiếp từ Window object
if Window.Gui then
    FluentFrame = Window.Gui
elseif Window.GUI then
    FluentFrame = Window.GUI
elseif Window.Main then
    FluentFrame = Window.Main
end

-- Nếu vẫn không có, duyệt toàn bộ để tìm
if not FluentFrame then
    for _, v in pairs(getgc(true)) do
        if type(v) == "table" then
            if v.Gui and typeof(v.Gui) == "Instance" then
                FluentFrame = v.Gui
                break
            end
        end
    end
end

-- Backup cuối: tìm trong tất cả service
if not FluentFrame then
    local services = {
        game:GetService("CoreGui"),
        game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")
    }
    for _, svc in ipairs(services) do
        local ok, children = pcall(function() return svc:GetChildren() end)
        if ok then
            for _, gui in ipairs(children) do
                if gui:IsA("ScreenGui") then
                    FluentFrame = gui
                    break
                end
            end
        end
        if FluentFrame then break end
    end
end

-- ========================================
-- NÚT TRÒN TRẮNG
-- ========================================
local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local PlayerGui = Players.LocalPlayer:WaitForChild("PlayerGui")

local ToggleGui = Instance.new("ScreenGui")
ToggleGui.Name = "PDZToggleButton"
ToggleGui.ResetOnSpawn = false
ToggleGui.DisplayOrder = 999999
ToggleGui.IgnoreGuiInset = true
ToggleGui.Parent = PlayerGui

local Button = Instance.new("ImageButton")
Button.Size = UDim2.fromOffset(56, 56)
Button.Position = UDim2.fromOffset(20, 300)
Button.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
Button.BorderSizePixel = 0
Button.AutoButtonColor = false
Button.ZIndex = 10
Button.Parent = ToggleGui

Instance.new("UICorner", Button).CornerRadius = UDim.new(1, 0)

local Stroke = Instance.new("UIStroke", Button)
Stroke.Color = Color3.fromRGB(160, 160, 160)
Stroke.Thickness = 2

local Label = Instance.new("TextLabel", Button)
Label.Size = UDim2.fromScale(1, 1)
Label.BackgroundTransparency = 1
Label.Text = "UI"
Label.TextColor3 = Color3.fromRGB(20, 20, 20)
Label.TextScaled = true
Label.Font = Enum.Font.GothamBold
Label.ZIndex = 11

-- ========================================
-- KÉO NÚT
-- ========================================
local dragging = false
local dragStartInput = nil
local dragStartPos = nil
local totalDrag = 0

Button.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        totalDrag = 0
        dragStartInput = input.Position
        dragStartPos = Button.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStartInput
        totalDrag = delta.Magnitude
        local screen = workspace.CurrentCamera.ViewportSize
        Button.Position = UDim2.fromOffset(
            math.clamp(dragStartPos.X.Offset + delta.X, 0, screen.X - 56),
            math.clamp(dragStartPos.Y.Offset + delta.Y, 0, screen.Y - 56)
        )
    end
end)

Button.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

-- ========================================
-- BẬT / TẮT UI
-- ========================================
local uiVisible = true

Button.InputEnded:Connect(function(input)
    if (input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1)
    and totalDrag < 10 then

        uiVisible = not uiVisible

        if FluentFrame then
            -- Ẩn/hiện ScreenGui của Fluent
            if FluentFrame:IsA("ScreenGui") then
                FluentFrame.Enabled = uiVisible
            else
                -- Nếu là Frame thì dùng Visible
                FluentFrame.Visible = uiVisible
            end
        else
            -- Không tìm thấy FluentFrame → dùng MinimizeKey giả lập
            -- Fluent có hàm Toggle nội bộ
            if Window.MinimizeFrame then
                Window.MinimizeFrame.Visible = uiVisible
            end
            -- Tắt tất cả ScreenGui không phải nút của mình
            for _, gui in ipairs(PlayerGui:GetChildren()) do
                if gui:IsA("ScreenGui") and gui.Name ~= "PDZToggleButton" then
                    gui.Enabled = uiVisible
                end
            end
        end

        if uiVisible then
            Button.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
            Label.TextColor3 = Color3.fromRGB(20, 20, 20)
        else
            Button.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
            Label.TextColor3 = Color3.fromRGB(210, 210, 210)
        end
    end
end)

Fluent:Notify({
    Title = "PDZ Hub",
    Content = "Load xong!.",
    Duration = 4
})
