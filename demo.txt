local SaveManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/master/Addons/SaveManager.lua"))()
local InterfaceManager = loadstring(game:HttpGet("https://raw.githubusercontent.com/dawid-scripts/Fluent/master/Addons/InterfaceManager.lua"))()

-- Tạo cửa sổ chính
local Window = Fluent:CreateWindow({
    Title = "PDZ Hub",
    SubTitle = "not share",
    TabWidth = 160,
    Size = UDim2.fromOffset(400, 309),
    Acrylic = true,
    Theme = "Dark",
    MinimizeKey = Enum.KeyCode.LeftControl
})

-- ==================== NÚT TOGGLE HÌNH TRÒN ====================

local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local RunService = game:GetService("RunService")

local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

-- Tạo ScreenGui chứa nút
local ToggleGui = Instance.new("ScreenGui")
ToggleGui.Name = "PDZToggleButton"
ToggleGui.ResetOnSpawn = false
ToggleGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
ToggleGui.DisplayOrder = 999
ToggleGui.Parent = PlayerGui

-- Nút hình tròn màu trắng
local Button = Instance.new("ImageButton")
Button.Name = "ToggleBtn"
Button.Size = UDim2.fromOffset(52, 52)
Button.Position = UDim2.new(0, 20, 0.5, 0)
Button.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
Button.BorderSizePixel = 0
Button.Image = ""
Button.AutoButtonColor = false
Button.Parent = ToggleGui

-- Bo tròn nút
local Corner = Instance.new("UICorner")
Corner.CornerRadius = UDim.new(1, 0)
Corner.Parent = Button

-- Đổ bóng nhẹ
local Stroke = Instance.new("UIStroke")
Stroke.Color = Color3.fromRGB(180, 180, 180)
Stroke.Thickness = 2
Stroke.Parent = Button

-- Chữ "pdz" ở giữa nút
local Label = Instance.new("TextLabel")
Label.Size = UDim2.fromScale(1, 1)
Label.BackgroundTransparency = 1
Label.Text = "UI"
Label.TextColor3 = Color3.fromRGB(30, 30, 30)
Label.TextScaled = true
Label.Font = Enum.Font.GothamBold
Label.Parent = Button

-- Padding chữ cho đẹp
local Padding = Instance.new("UIPadding")
Padding.PaddingTop = UDim.new(0, 10)
Padding.PaddingBottom = UDim.new(0, 10)
Padding.PaddingLeft = UDim.new(0, 8)
Padding.PaddingRight = UDim.new(0, 8)
Padding.Parent = Label


-- ====== KÉO NÚT ======

local dragging = false
local dragStartInput
local dragStartPos

Button.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStartInput = input.Position
        dragStartPos = Button.Position
    end
end)

Button.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging then
        if input.UserInputType == Enum.UserInputType.Touch
        or input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStartInput
            local screen = workspace.CurrentCamera.ViewportSize
            local newX = math.clamp(dragStartPos.X.Offset + delta.X, 0, screen.X - 52)
            local newY = math.clamp(dragStartPos.Y.Offset + delta.Y, 0, screen.Y - 52)
            Button.Position = UDim2.fromOffset(newX, newY)
        end
    end
end)

-- ====== BẬT TẮT UI ======

local uiVisible = true
local movedDist = 0
local inputStart

Button.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1 then
        inputStart = input.Position
        movedDist = 0
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if inputStart and (input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseMovement) then
        movedDist = (input.Position - inputStart).Magnitude
    end
end)

Button.InputEnded:Connect(function(input)
    if (input.UserInputType == Enum.UserInputType.Touch
    or input.UserInputType == Enum.UserInputType.MouseButton1)
    and movedDist < 8 then

        uiVisible = not uiVisible

        -- Tìm GUI Fluent và ẩn/hiện
        for _, gui in ipairs(PlayerGui:GetChildren()) do
            if gui:IsA("ScreenGui") and gui.Name ~= "PDZToggleButton" then
                gui.Enabled = uiVisible
            end
        end

        if uiVisible then
            Button.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
            Label.TextColor3 = Color3.fromRGB(30, 30, 30)
        else
            Button.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
            Label.TextColor3 = Color3.fromRGB(200, 200, 200)
        end
    end
end)

Fluent:Notify({
    Title = "PDZ Hub",
    Content = "Load xong! Nhấn nút tròn để bật/tắt UI",
    Duration = 4
})
