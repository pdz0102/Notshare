--// FLUENT UI LIBRARY

local Fluent = loadstring(game:HttpGet(
    "https://github.com/dawid-scripts/Fluent/releases/latest/download/main.lua"
))()

    "https://raw.githubusercontent.com/NevaHub0/SaveManager/main/.lua"
))()

    "https://raw.githubusercontent.com/NevaHub0/InterfaceManager/main/.lua"
))()

-- WINDOW
--====================================================

})

Tabs.Misc:AddSlider("WalkSpeed", {
    Title = "WalkSpeed",
    Description = "Player Speed",

    Min = 16,
    Max = 100,
    Rounding = 1,

    Callback = function(Value)

        game.Players.LocalPlayer.Character.Humanoid.WalkSpeed = Value

    end
})

    Title = "Jump Power",

    Default = 50,
    Min = 50,
    Max = 200,
    Rounding = 1,
